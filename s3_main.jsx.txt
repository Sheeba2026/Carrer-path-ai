<script>
  function toggleComparisonView() {
    const drawer = document.getElementById('comparison-drawer');
    const text = document.getElementById('comparison-btn-text');
    if (!drawer) return;
    const isHidden = drawer.classList.contains('hidden');
    if (isHidden) {
      drawer.classList.remove('hidden');
      if (text) text.textContent = 'Hide comparison matrix';
    } else {
      drawer.classList.add('hidden');
      if (text) text.textContent = 'Compare roles side-by-side';
    }
  }

  function toggleAdjacentRoles() {
    const drawer = document.getElementById('adjacent-roles-drawer');
    if (!drawer) return;
    drawer.classList.toggle('hidden');
  }
</script></main>
