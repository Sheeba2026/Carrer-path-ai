<script>
  // Simple Interactive State Management for Step 2
  function updateSkillCounts() {
    const items = document.querySelectorAll('.skill-vector-item');
    const totalCount = items.length;
    let transferableCount = 0;
    let technicalCount = 0;

    items.forEach(item => {
      const cat = item.getAttribute('data-category');
      if (cat === 'technical') {
        technicalCount++;
      } else {
        transferableCount++;
      }
    });

    // Update Counter Badges
    const counterBadge = document.getElementById('active-skill-counter');
    if (counterBadge) counterBadge.textContent = totalCount + ' Total';

    const statTotal = document.getElementById('stat-total-skills');
    if (statTotal) statTotal.textContent = totalCount + ' skills';

    const statRatio = document.getElementById('stat-skill-ratio');
    if (statRatio) statRatio.textContent = `${transferableCount} Transferable / ${technicalCount} Tech`;

    const previewTeaser = document.getElementById('preview-teaser-text');
    if (previewTeaser) {
      previewTeaser.textContent = `Next: We'll match these ${totalCount} verified skills against real-world roles across analytics, operations, and coordination.`;
    }
  }

  function removeSkillCard(btn) {
    const card = btn.closest('.skill-vector-item');
    if (card) {
      card.style.opacity = '0';
      card.style.transform = 'scale(0.95)';
      card.style.transition = 'all 0.2s ease-out';
      setTimeout(() => {
        card.remove();
        updateSkillCounts();
      }, 200);
    }
  }

  function addNewSkillFromInput() {
    const input = document.getElementById('new-skill-input');
    if (!input || !input.value.trim()) return;
    const skillName = input.value.trim();
    createSkillCard(skillName, 'transferable', 'User added manual skill capability');
    input.value = '';
    updateSkillCounts();
  }

  function quickAddSkill(skillName, category) {
    createSkillCard(skillName, category, 'User quick-added from domain suggestions');
    updateSkillCounts();
  }

  function createSkillCard(name, category, snippet) {
    const container = document.getElementById('skills-cloud-grid');
    if (!container) return;

    const isTech = category === 'technical';
    const tagColor = isTech ? 'bg-primary-container text-primary' : 'bg-tertiary text-tertiary';
    const tagLabel = isTech ? 'Technical' : 'Transferable';
    const dotColor = isTech ? 'bg-primary-container' : 'bg-tertiary';

    const card = document.createElement('div');
    card.className = 'skill-vector-item flex flex-col justify-between p-4 rounded-xl bg-surface-container hover:bg-surface-container-high transition-all shadow-sm group relative animate-in fade-in zoom-in-95 duration-200';
    card.setAttribute('data-category', category);

    card.innerHTML = `
      <div className="flex items-start justify-between gap-3">
        <div className="space-y-1 min-w-0">
          <div className="flex items-center gap-2">
            <span className="w-2 h-2 rounded-full ${dotColor}"></span>
            <span className="font-label-sm text-label-sm uppercase tracking-wide ${tagColor.split(' ')[1]}">${tagLabel}</span>
          </div>
          <h3 className="font-label-lg text-label-lg text-on-surface font-bold group-hover:text-primary transition-colors">
            ${escapeHtml(name)}
          </h3>
        </div>
        <button type="button"  title="Remove skill" className="text-outline hover:text-error transition-colors p-1 rounded hover:bg-surface-container-highest">
          <span className="material-symbols-outlined text-[18px]">close</span>
        </button>
      </div>
      <div className="mt-3.5 pt-3 bg-surface-container-lowest/40 rounded-lg p-2.5 flex items-start gap-2">
        <span className="material-symbols-outlined text-[15px] text-outline shrink-0 mt-0.5">add_task</span>
        <p className="font-body-sm text-body-sm text-on-surface-variant/90 italic line-clamp-2">
          &ldquo;${escapeHtml(snippet)}&rdquo;
        </p>
      </div>
    `;

    container.appendChild(card);
  }

  function escapeHtml(string) {
    return String(string).replace(/[&<>"'`=\/]/g, function (s) {
      return ({
        '&': '&amp;',
        '<': '&lt;',
        '>': '&gt;',
        '"': '&quot;',
        "'": '&#39;',
        '/': '&#x2F;',
        '`': '&#x60;',
        '=': '&#x3D;'
      })[s];
    });
  }

  function goToStep3() {
    if (typeof goToStage === 'function') {
      goToStage(3);
    } else {
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }
  }

  // Handle enter key on input
  document.addEventListener('DOMContentLoaded', () => {
    const input = document.getElementById('new-skill-input');
    if (input) {
      input.addEventListener('keydown', (e) => {
        if (e.key === 'Enter') {
          e.preventDefault();
          addNewSkillFromInput();
        }
      });
    }
  });
