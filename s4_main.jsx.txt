<script>
    const completedMilestones = { 1: false, 2: false, 3: false };

    function handleMilestoneToggle(step, checkbox) {
      completedMilestones[step] = checkbox.checked;
      
      const badge = document.getElementById(`m${step}-badge-status`);
      const card = document.getElementById(`milestone-card-${step}`);
      
      if (checkbox.checked) {
        if (badge) {
          badge.className = 'inline-flex items-center gap-2 px-3 py-1 rounded-full bg-tertiary-container text-on-tertiary font-label-sm text-label-sm font-semibold';
          badge.innerHTML = '<span className="material-symbols-outlined text-[14px]" style={{"fontVariationSettings":"\\'FILL\\' 1"}}>check_circle</span> Complete (Logged)';
        }
        if (card) {
          card.classList.add('bg-surface-container');
          card.classList.remove('bg-surface-container-low');
        }
      } else {
        if (badge) {
          badge.className = 'inline-flex items-center gap-2 px-3 py-1 rounded-full bg-surface-variant text-on-surface-variant font-label-sm text-label-sm';
          badge.innerHTML = '<span className="w-1.5 h-1.5 rounded-full bg-outline"></span> Incomplete (Pending Action)';
        }
        if (card) {
          card.classList.remove('bg-surface-container');
          card.classList.add('bg-surface-container-low');
        }
      }

      updateReadoutProgress();
    }

    function updateReadoutProgress() {
      const checkedCount = Object.values(completedMilestones).filter(Boolean).length;
      const baseReadiness = 70; // baseline affinity
      const additionalPerStep = 10; // max reaches 100%
      const newScore = Math.min(100, baseReadiness + (checkedCount * additionalPerStep));

      // Update counters
      const counterEl = document.getElementById('tracker-counter');
      if (counterEl) {
        counterEl.textContent = `${checkedCount}/3`;
      }

      const subtextEl = document.getElementById('milestones-status-subtext');
      if (subtextEl) {
        const remaining = 3 - checkedCount;
        subtextEl.textContent = remaining === 0 ? 'All milestones fulfilled!' : `${remaining} actions remaining`;
      }

      const labelEl = document.getElementById('progress-percentage-label');
      if (labelEl) {
        labelEl.textContent = `${newScore}%`;
      }

      // Update SVG circular bar
      const svgProgress = document.getElementById('svg-progress-bar');
      if (svgProgress) {
        const pct = (checkedCount / 3) * 100;
        svgProgress.setAttribute('stroke-dasharray', `${pct}, 100`);
      }
    }

    function handleExportPlan() {
      const checkedCount = Object.values(completedMilestones).filter(Boolean).length;
      const exportMsg = `Learning Plan Export initiated.\nPathway: Project Coordinator & Operations Lead\nProgress: ${checkedCount} of 3 milestones marked completed.\n\nFormatting structured template for Notion & PDF...`;
      alert(exportMsg);
    }

    function handleShareWithMentor() {
      const shareUrl = window.location.href;
      if (navigator.clipboard) {
        navigator.clipboard.writeText(shareUrl).then(() => {
          alert("Sharable mentor review link copied to clipboard!\nSend this URL to your advisor to review your project milestones.");
        });
      } else {
        alert("Sharable link: " + shareUrl);
      }
    }
  </script>
